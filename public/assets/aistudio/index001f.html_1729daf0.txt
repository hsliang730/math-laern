        function evaluatePasswordStrength(password) {
            if (!password) return { score: 0, text: '尚未輸入', color: 'bg-gray-200', valid: false };
            if (password.length < 8) return { score: 20, text: '太短（至少需要 8 碼）', color: 'bg-red-500', valid: false };

            const sequential = /(012|123|234|345|456|567|678|789|890|abc|bcd|cde|def|efg|fgh|ghi|hij|ijk|jkl|klm|lmn|mno|nop|opq|pqr|qrs|rst|stu|tuv|uvw|vwx|wxy|xyz|qwert|asdfg|zxcvb)/i;
            const repeated = /(.)\1{3,}/; 

            if (sequential.test(password)) return { score: 30, text: '包含連續字元，不安全', color: 'bg-red-500', valid: false };
            if (repeated.test(password)) return { score: 30, text: '包含重複字元，不安全', color: 'bg-red-500', valid: false };

            let score = 40;
            if (/[a-z]/.test(password)) score += 20;
            if (/[A-Z]/.test(password)) score += 20;
            if (/[0-9]/.test(password)) score += 10;
            if (/[^A-Za-z0-9]/.test(password)) score += 10;

            if (score < 70) return { score: 50, text: '強度不足（請混合大小寫英文與數字）', color: 'bg-orange-500', valid: false };
            else if (score < 90) return { score: 80, text: '強度良好', color: 'bg-blue-500', valid: true };
            else return { score: 100, text: '強度極佳', color: 'bg-green-500', valid: true };
        }

        async function handleAuth() {
            const userInput = document.getElementById('username').value.trim();
            const password = document.getElementById('password').value;
            const rememberMe = document.getElementById('remember-me').checked;
            const msgEl = document.getElementById('message');
            msgEl.innerText = '';

            if (!userInput || !password) {
                msgEl.innerText = '請完整填寫帳號與密碼';
                triggerErrorAnimation();
                return;
            }

            if (rememberMe) {
                localStorage.setItem('remember_username', userInput);
                localStorage.setItem('remember_password', password);
            } else {
                localStorage.removeItem('remember_username');
                localStorage.removeItem('remember_password');
            }

            const email = userInput + '@mathlearn.com';

            if (isSignUp) {
                const { error } = await supabaseClient.auth.signUp({ 
                    email, password, options: { data: { must_change_password: true, nickname: '' } }
                });
                if (error) {
                    msgEl.innerText = '註冊失敗：' + error.message;
                    triggerErrorAnimation();
                } else {
                    triggerSuccessAnimation(() => {
                        alert('註冊成功！請直接登入');
                        toggleMode();
                    });
                }
            } else {
                const { data: banCheck } = await supabaseClient.from('site_config').select('value').eq('key', `banned_user_${userInput}`).single();
                if (banCheck && banCheck.value === 'true') {
                    msgEl.innerText = '❌ 您的帳號已被管理員封鎖或禁言，無法登入。';
                    triggerErrorAnimation();
                    return;
                }

                const { data, error } = await supabaseClient.auth.signInWithPassword({ email, password });
                if (error) {
                    msgEl.innerText = '登入失敗：帳號或密碼錯誤';
                    triggerErrorAnimation();
                } else {
                    currentUserObject = data.user;
                    currentUsername = userInput;

                    const { data: forcePwCheck } = await supabaseClient.from('site_config').select('value').eq('key', `force_pw_${userInput}`).single();
                    const mustChange = (forcePwCheck && forcePwCheck.value === 'true') || password === '00000000' || (currentUserObject.user_metadata && currentUserObject.user_metadata.must_change_password);

                    if (mustChange) {
                        triggerSuccessAnimation(() => {
                            document.getElementById('auth-card').classList.add('hidden');
                            document.getElementById('change-password-card').classList.remove('hidden');
                        });
                        return;
                    }

                    recordUserLogin(userInput);
                    triggerSuccessAnimation(() => {
                        showDashboard(userInput);
                    });
                }
            }
        }

        async function handleForceChangePassword() {
            const newPw = document.getElementById('new-password').value;
            const confirmPw = document.getElementById('confirm-password').value;
            const msg = document.getElementById('pw-message');
            msg.innerText = '';

            if (!newPw || !confirmPw) {
                msg.innerText = '請輸入新密碼';
                return;
            }

            if (newPw !== confirmPw) {
                msg.innerText = '兩次輸入的密碼不一致';
                return;
            }

            const strength = evaluatePasswordStrength(newPw);
            if (!strength.valid) {
                msg.innerText = '密碼強度不足：' + strength.text;
                return;
            }

            const { error } = await supabaseClient.auth.updateUser({ password: newPw, data: { must_change_password: false } });
            if (error) {
                msg.innerText = '密碼變更失敗：' + error.message;
            } else {
                await supabaseClient.from('site_config').delete().eq('key', `force_pw_${currentUsername}`);
                alert('密碼修改成功！即將為您進入系統。');
                recordUserLogin(currentUsername);
                showDashboard(currentUsername);
            }
        }

        function startSessionTimer() {
            stopSessionTimer();
            sessionSeconds = 0;
            const timerEl = document.getElementById('session-timer');
            sessionTimerInterval = setInterval(() => {
                sessionSeconds++;
                const mins = Math.floor(sessionSeconds / 60);
                const secs = sessionSeconds % 60;
                if (timerEl) timerEl.innerText = `使用 ${mins}分${secs}秒`;
            }, 1000);
        }

            if (sessionTimerInterval) clearInterval(sessionTimerInterval);

        async function uploadHandoutRealFile() {
            const title = document.getElementById('handout-title').value.trim();
            const fileInput = document.getElementById('handout-file-input');
            const file = fileInput.files[0];

            if (!title || !file) return alert('請填寫檔案名稱並選擇檔案');

            if (file.size > 8 * 1024 * 1024) {
                return alert('⚠️ 檔案容量過大！為保護系統穩定度，檔案上限為 8MB。');
            }

            try {
                const fileExt = file.name.split('.').pop();
                const fileName = `${Date.now()}_${Math.random().toString(36).substring(2, 7)}.${fileExt}`;

                const { data: uploadData, error: uploadError } = await supabaseClient.storage.from('handouts').upload(fileName, file);

                let finalUrl = '';
                if (!uploadError && uploadData) {
                    const { data: publicUrlData } = supabaseClient.storage.from('handouts').getPublicUrl(fileName);
                    finalUrl = publicUrlData.publicUrl;
                } else {
                    finalUrl = await new Promise((resolve) => {
                        const reader = new FileReader();
                        reader.onload = (e) => resolve(e.target.result);
                        reader.readAsDataURL(file);
                    });
                }

                await supabaseClient.from('handouts').insert([{
                    title,
                    url: finalUrl,
                    is_folder: false,
                    parent_id: currentFolderId
                }]);

                document.getElementById('handout-title').value = '';
                fileInput.value = '';
                alert('檔案上傳成功！');
                renderFolderView();
            } catch (err) {
                alert('上傳失敗：' + err.message);
            }
        }

        async function loadAdminImpersonateUsers() {

                const { data: activeLogins } = await supabaseClient.from('site_config').select('*').like('key', 'active_login_%');
                const { data: bans } = await supabaseClient.from('site_config').select('*').like('key', 'banned_user_%');

                const banMap = {};
                if (bans) bans.forEach(b => banMap[b.key.replace('banned_user_', '')] = (b.value === 'true'));

                const forcePwMap = {};
                if (forcePws) forcePws.forEach(f => forcePwMap[f.key.replace('force_pw_', '')] = (f.value === 'true'));

                        </td>
                    </tr>`;

        // --- PDF 預覽 modal & zoom ---
        function openFilePreview(url, title) {
            currentFileUrl = url;
            currentFileName = title;
            currentZoom = 100;
            document.getElementById('preview-modal-title').innerText = title;
            document.getElementById('file-preview-modal').classList.remove('hidden');

            const container = document.getElementById('pdf-pages-container');
            const imgEl = document.getElementById('preview-img');
            const loadingEl = document.getElementById('preview-loading');

            container.innerHTML = '';
            imgEl.classList.add('hidden');
            loadingEl.classList.remove('hidden');

            if (url.startsWith('data:image/') || url.match(/\.(jpeg|jpg|gif|png|webp)($|\?)/i)) {
                imgEl.src = url;
                imgEl.style.transform = `scale(${currentZoom / 100})`;
                imgEl.classList.remove('hidden');
                loadingEl.classList.add('hidden');
            } else {
                pdfjsLib.getDocument(url).promise.then(pdf => {
                    currentPdfDoc = pdf;
                    renderAllPdfPages(pdf);
                    loadingEl.classList.add('hidden');
                }).catch(err => {
                    console.error("PDF 載入失敗:", err);
                    loadingEl.innerHTML = `<span class="text-xs text-red-500">無法預覽此檔案，您可以點擊右上角「下載」直接下載查看。</span>`;
                });
            }
        }

        async function renderAllPdfPages(pdf) {
            const container = document.getElementById('pdf-pages-container');
            container.innerHTML = '';
            for (let pageNum = 1; pageNum <= pdf.numPages; pageNum++) {
                const page = await pdf.getPage(pageNum);
                const scale = (currentZoom / 100) * 1.5;
                const viewport = page.getViewport({ scale });

                const canvas = document.createElement('canvas');
                canvas.className = 'shadow-md rounded mb-4 max-w-full';
                const context = canvas.getContext('2d');
                canvas.height = viewport.height;
                canvas.width = viewport.width;

                const renderContext = { canvasContext: context, viewport };
                await page.render(renderContext).promise;
                container.appendChild(canvas);
            }
        }

        function zoomIn() {
            if (currentZoom < 250) {
                currentZoom += 25;
                updateZoomView();
            }
        }

        function zoomOut() {
            if (currentZoom > 50) {
                currentZoom -= 25;
                updateZoomView();
            }
        }

        function zoomReset() {
            currentZoom = 100;
            updateZoomView();
        }

        function updateZoomView() {
            document.getElementById('zoom-level-text').innerText = `${currentZoom}%`;
            if (currentPdfDoc) {
                renderAllPdfPages(currentPdfDoc);
            } else {
                const imgEl = document.getElementById('preview-img');
                if (imgEl) imgEl.style.transform = `scale(${currentZoom / 100})`;
            }
        }

        function closeFilePreview() {
            document.getElementById('file-preview-modal').classList.add('hidden');
            currentPdfDoc = null;
            currentFileUrl = '';
        }

        function downloadCurrentFile() {
            if (!currentFileUrl) return;
            const a = document.createElement('a');
            a.href = currentFileUrl;
            a.download = currentFileName || 'download';
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
        }
